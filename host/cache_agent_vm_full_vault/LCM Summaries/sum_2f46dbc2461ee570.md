    local dur="$1"

    local unit="${dur: -1}"
    local seconds=0
    case "$unit" in
        h) seconds=$((num * 3600)) ;;
        d) seconds=$((num * 86400)) ;;
        m) seconds=$((num * 60)) ;;

    esac
