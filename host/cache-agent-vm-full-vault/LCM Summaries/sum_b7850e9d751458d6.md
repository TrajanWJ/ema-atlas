# Sets: EXTENDS_KEY, EXPIRES_DURATION
parse_flags() {
    EXTENDS_KEY=""
    EXPIRES_DURATION=""
    local new_args=()
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --extends)
                EXTENDS_KEY="$2"
                shift 2
                ;;
            --expires)
                EXPIRES_DURATION="$2"
                shift 2
                ;;
            *)
                new_args+=("$1")
                shift

        esac
    done
    PARSED_ARGS=("${new_args[@]}")
}

# Convert duration string (24h, 7d, 30m) to epoch timestamp
expires_to_epoch() {
    local dur="$1"
    local num="${dur%[hdm]}"
    local unit="${dur: -1}"
    local seconds=0
    case "$unit" in
        h) seconds=$((num * 3600)) ;;
        d) seconds=$((num * 86400)) ;;
        m) seconds=$((num * 60)) ;;
        *) echo "Invalid duration unit: $unit (use h/d/m)" >&2; return 1 ;;
    esac
    echo $(($(date +%s) + seconds))
}
