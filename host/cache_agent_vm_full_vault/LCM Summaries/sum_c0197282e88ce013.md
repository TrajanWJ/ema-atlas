function _renderTodaysPlan() {
  const d = _lifeData;
  const blocks = d?.schedule || [];
  const plan = d?.dailyPlan;
  const now = new Date();
  const hours = Array.from({length: 16}, (_, i) => i + 6); // 6am to 10pm

  let timelineHtml = hours.map(h => {
    const hourStr = h.toString().padStart(2, '0');
    const label = h > 12 ? `${h-12}p` : h === 12 ? '12p' : `${h}a`;
    const block = blocks.find(b => {
      const bh = new Date(b.start).getUTCHours();
      return bh === h;
    });
    const isCurrent = now.getUTCHours() === h;
    const blockClass = block ? ` has-block${block.status === 'completed' ? ' completed' : ''}` : '';
    const currentClass = isCurrent ? ' current-hour' : '';

    return `<div class="timeline-hour${blockClass}${currentClass}" title="${block ? block.title : ''}">
      <span class="hour-label">${label}</span>
      ${block ? `<div class="hour-block">${_escHtml(block.title)}</div>` : ''}
    </div>`;
  }).join('');

  const intentionsHtml = plan?.intentions?.length
    ? plan.intentions.map(i => `<div class="plan-intention">• ${_escHtml(i)}</div>`).join('')
    : '<div class="plan-intention dim">No morning intentions set</div>';

  return `
    <div class="widget-header">
      <span class="widget-icon">📋</span>
      <span class="widget-title">Today's Plan</span>
      <span class="widget-badge">${blocks.length} blocks</span>
    </div>
    <div class="plan-intentions">
      <div class="plan-section-label">Morning Intentions</div>
      ${intentionsHtml}
    </div>
    <div class="plan-timeline">${timelineHtml}</div>
  `;
}
