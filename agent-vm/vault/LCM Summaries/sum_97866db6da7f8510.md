  if (!messages || !messages.length) {
    container.innerHTML = '<div style="padding:40px;text-align:center;color:#6c7086">No messages in this channel</div>';
    return;
  }
  const sorted = [...messages].reverse(); // API returns newest first
  container.innerHTML = sorted.map(msg => {
    const author = msg.author || {};
    const time = new Date(msg.timestamp || Date.now()).toLocaleTimeString([], {hour:'2-digit',minute:'2-digit'});

    return `<div class="talk-msg" style="display:flex;gap:10px;padding:8px 16px;border-bottom:1px solid #181825">

      <div style="flex:1;min-width:0">
        <div><span style="font-weight:600;font-size:13px;color:#cdd6f4">${author.username||'Unknown'}</span>${isBot?'<span style="background:#5865f2;color:white;font-size:9px;padding:1px 4px;border-radius:3px;margin-left:6px">BOT</span>':''} <span style="font-size:11px;color:#6c7086">${time}</span></div>

      </div>
    </div>`;
  }).join('');
  container.scrollTop = container.scrollHeight;
}
