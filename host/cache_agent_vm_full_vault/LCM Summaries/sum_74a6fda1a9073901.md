  /** Build full context string to prepend to Claude's input */
  buildSessionContext(session: SessionRecord, userMessage: string): string {
    const dir = session.directory;
    const projectName = session.projectName || basename(dir);

    const projectDocs = this.readProjectDocs(dir);
    const stack = this.detectStack(dir);
    const git = this.readGitState(dir);
    const history = this.readHistory(session.id);
    const prefs = this.readPreferences();
    const fileMentions = this.detectFileMentions(userMessage, dir);

    // Build context block
    let contextBlock = `[CONTEXT]\nProject: ${projectName} at ${dir}\n`;
    contextBlock += projectDocs + "\n";

    if (stack.length > 0) {
      contextBlock += `\nStack: ${stack.join(", ")}\n`;
    }

    if (git) {
      contextBlock += `Git: branch=${git.branch} | ${git.uncommitted} uncommitted changes\n`;
      if (git.commits.length > 0) {
        contextBlock += "Recent commits:\n";
        for (const c of git.commits) {
          contextBlock += `- ${c}\n`;
        }
      }
    }

    if (fileMentions.length > 0) {
      contextBlock += `\nMentioned files: ${fileMentions.join(", ")}\n`;
    }

    if (prefs) {
      contextBlock += `\nUser preferences:\n${prefs}\n`;
    }

    contextBlock += "[/CONTEXT]";

    // Build history block
    let historyBlock = "";
    if (history.length > 0) {
      historyBlock = "\n\n[HISTORY]\n";
      for (const msg of history) {
        const label = msg.role === "user" ? "User" : "Assistant";
        const content =
          msg.content.length > MAX_HISTORY_RESPONSE
            ? msg.content.slice(0, MAX_HISTORY_RESPONSE) + "..."
            : msg.content;
        historyBlock += `${label}: ${content}\n`;
      }
      historyBlock += "[/HISTORY]";
    }

    // Combine and enforce total size limit
    let full = contextBlock + historyBlock + "\n\n" + userMessage;

    if (full.length > MAX_CONTEXT_SIZE) {
      // Truncate history first, then project docs
      const overhead = full.length - MAX_CONTEXT_SIZE;
      if (historyBlock.length > overhead) {
        historyBlock = historyBlock.slice(0, historyBlock.length - overhead - 20) + "\n...\n[/HISTORY]";
      } else {
        historyBlock = "";
        const remaining = MAX_CONTEXT_SIZE - userMessage.length - 100;
        if (contextBlock.length > remaining) {
          contextBlock = contextBlock.slice(0, remaining) + "\n[/CONTEXT]";
        }
      }
      full = contextBlock + historyBlock + "\n\n" + userMessage;
    }

    return full;
  }
